##STR. FUNCTION##

upper()	Converts to uppercase "hello".upper() → "HELLO"

lower()	Converts to lowercase	"HELLO".lower() → "hello"

title()	Capitalizes each word "python notes".title() → "Python Notes"

strip()	Removes whitespace	" hi ".strip() → "hi"

replace(a,b)	Replaces substring	"apple".replace("a","A") → "Apple"

split()	Splits into list	"a,b,c".split(",") → ['a','b','c']

join()	Joins list into string	",".join(['a','b']) → "a,b"
startswith()	Checks prefix

	"python".startswith("py") → True

endswith()	Checks suffix	"notes.txt".endswith(".txt") → True

find()	Finds index of substring	"banana".find("na") → 2

count()	Counts occurrences	"banana".count("a") → 3
